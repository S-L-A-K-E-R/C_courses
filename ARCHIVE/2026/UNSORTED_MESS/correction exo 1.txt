#include <stdio.h>
#include <string.h>
#define NB_CARAC_MAX 100

int compterOccurs(char string[], char caractere) {
    int i = 0;
    int compteur = 0;
    for(i = 0; string[i] != '\0'; i++) {
        if(string[i] == caractere) {
            compteur++;
        }
    }
    return compteur;
}

void saisirPhrase(char str[], int taillePhysique) {

    fgets(str, taillePhysique, stdin); // stdin = source clavier
    if (str[strlen(str)-1] == '\n') { // si le dernier caractere est un retour à la ligne
        str[strlen(str)-1] = '\0';
    }
    fflush(stdin); // au cas où la taille physique ait été atteinte, il reste la suite dans le buffer qu'il faut vider.

}
